\section{Scale Expansion}

\begin{itemize}
    \item Increase capacity utilization from 50\% (Year 1) to 85\% by Year 6.
    \item Explore exports to Middle East, EU (Germany, France), and ASEAN markets.
\end{itemize}

\section{Product Diversification}

\begin{itemize}
    \item Biodegradable cutlery, straws, cling films.
    \item Integration with blockchain for supply chain transparency in eco-certifications.
\end{itemize}

\section{Circular Economy Alliances}

\begin{itemize}
    \item Collaborations with agri-produce processors (e.g., sugarcane waste suppliers).
    \item Pilot carbon credit-linked production via carbon-negative feedstocks (e.g., methane).
    \item \textbf{\faFlask\ \ MIT \& ETH Zurich researchers are working on methane-fed PHA (polyhydroxyalkanoates) plastics} – a promising future substitute to PLA. \\
    \textit{(Source: Nature Sustainability, 2023)}
\end{itemize}
