\section{Goal}

Production-hardened, multi-tenant--ready, Kubernetes-deployable platform with full observability.

\section{New Technology}

\begin{itemize}[leftmargin=*]
  \item \textbf{Kubernetes} (K3s)
  \item \textbf{Traefik} or \textbf{Kong} API gateway
  \item \textbf{Prometheus} and \textbf{Grafana}
  \item \textbf{Keycloak} or \textbf{NextAuth}
  \item \textbf{Redis Streams} or \textbf{RabbitMQ}
  \item \textbf{Qdrant} or \textbf{Weaviate}
\end{itemize}

\section{Features}

\begin{itemize}[leftmargin=*]
  \item Multi-tenant support
  \item RBAC roles: Admin, Investor, Operator, Auditor
  \item Real-time streaming chat (WebSocket)
  \item A/B testing
  \item Audit logging
  \item Automated certificate renewal workflow
  \item Webhook integrations (e.g., Slack)
  \item GraphQL API alongside REST
  \item Internationalization (English / Hindi)
\end{itemize}

\section{Always-Free or Low-Cost Components}

\begin{itemize}[leftmargin=*]
  \item \textbf{Cloudflare} CDN and DNS
  \item \textbf{Let's Encrypt} SSL
  \item \textbf{$\sim$\$10/year} \texttt{.in} domain
  \item \textbf{Cloudflare R2} or \textbf{Backblaze B2} (about 10\,GB free)
\end{itemize}

\section{Implementation Steps}

\begin{enumerate}[leftmargin=*]
  \item Create Kubernetes manifests
  \item Set up Helm charts
  \item Configure Traefik ingress
  \item Deploy Keycloak for authentication
  \item Add Prometheus and Grafana stack
  \item Implement the GraphQL layer
  \item Add i18n (EN / HI)
  \item Write Terraform IaC
  \item Set up feature flags with Unleash
\end{enumerate}

\section{Version Control}

\begin{itemize}[leftmargin=*]
  \item \textbf{GitFlow} or trunk-based development
  \item Feature flags via \textbf{Unleash}
  \item Helm charts and \textbf{Terraform} / \textbf{OpenTofu}
  \item Branch protection and signed commits
  \item Automated changelog generation
\end{itemize}
